    service pods(logos:kubernetes)[CI Job Pods] in eks
    service rds(logos:aws-rds)[RDS PostgreSQL] in eks
    service s3(logos:aws-s3)[S3 Logs] in eks

    github:R --> L:webhook
    webhook:R --> L:temporal
    temporal:R --> L:worker
    worker:B --> T:pods
    temporal:B --> T:rds
    worker:R --> L:s3`} />
